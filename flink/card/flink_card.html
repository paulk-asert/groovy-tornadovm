@GpuKernel
static void nearest(KernelContext ctx, FloatArray points, FloatArray centroids,
                    IntArray out, int n, int k, int d) {
    int i = ctx.globalIdx
    if (i < n) {
        int best = 0; float bestDist = 0.0f
        for (int c = 0; c < k; c++) {
            float dist = 0.0f
            for (int j = 0; j < d; j++) {
                float diff = points.get(i * d + j) - centroids.get(c * d + j)
                dist = dist + diff * diff
            }
            if (c == 0 || dist < bestDist) { bestDist = dist; best = c }
        }
        out.set(i, best)
    }
}
